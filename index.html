    <script>

        // === 게임 변수 ===

        let P1_SCORE = 0;

        let P2_SCORE = 0;

        let P1_POINTS = 99;

        let P2_POINTS = 99;

        let CURRENT_ROUND = 1;

        const MAX_ROUNDS = 9;

        const WIN_SCORE = 5;

        let currentPlayer = 1; // 1: P1, 2: P2

        let firstPlayer = 1; // 라운드 선 플레이어

        let p1Bet = 0;

        let p2Bet = 0;

        let isFirstBet = true; // 라운드 첫 번째 베팅인지 (P1 베팅)

        let isGameOver = false;

        // === DOM 요소 ===

        const $p1Status = document.getElementById('player1Status');

        const $p2Status = document.getElementById('player2Status');

        const $roundStatus = document.getElementById('roundStatus');

        const $turnInfo = document.getElementById('turnInfo');

        const $pointInput = document.getElementById('pointInput');

        const $betButton = document.getElementById('betButton');

        const $resultPopup = document.getElementById('resultPopup');

        const $resultHeader = document.getElementById('resultHeader');

        const $roundResults = document.getElementById('roundResults');

        const $passMessage = document.getElementById('passMessage');

        const $passButton = document.getElementById('passButton');

        const $nextRoundButton = document.getElementById('nextRoundButton');

        const $restartButton = document.getElementById('restartButton');

        const $p1PointsDisplay = document.getElementById('p1PointsDisplay');

        const $p2PointsDisplay = document.getElementById('p2PointsDisplay');

        // === 초기화 및 업데이트 함수 ===

        function updateUI() {

            // 1. 점수 및 라운드 업데이트

            $p1Status.textContent = `P1: ${P1_SCORE}점`;

            $p2Status.textContent = `P2: ${P2_SCORE}점`;

            $roundStatus.textContent = `라운드 ${CURRENT_ROUND}/${MAX_ROUNDS}`;

            // 2. 턴 정보 업데이트

            const currentPoints = (currentPlayer === 1) ? P1_POINTS : P2_POINTS;

            $turnInfo.textContent = `P${currentPlayer} 턴입니다. (${currentPoints}P 보유)`;

            $pointInput.max = currentPoints;

            if ($pointInput.value > currentPoints) {

                $pointInput.value = currentPoints; // 보유 포인트를 넘지 않도록

            }

            $pointInput.min = 1; // 최소 1점 베팅

            // 3. 턴 표시 하이라이트

            $p1Status.classList.toggle('current-turn', currentPlayer === 1);

            $p2Status.classList.toggle('current-turn', currentPlayer === 2);

            // 4. 포인트 표시등 업데이트

            updatePointLights(P1_POINTS, $p1PointsDisplay);

            updatePointLights(P2_POINTS, $p2PointsDisplay);

        }

        function updatePointLights(points, displayElement) {

            const lights = displayElement.querySelectorAll('.point-light');

            const step = Math.floor(points / 20); // 0-19(0), 20-39(1), ..., 80-99(4) -> 5단계

           

            for (let i = 0; i < 5; i++) {

                // 80-99 (4개), 60-79 (3개) ...

                if (i <= step) {

                    lights[i].classList.add('on');

                } else {

                    lights[i].classList.remove('on');

                }

            }

        }

        function initGame() {

            P1_SCORE = 0;

            P2_SCORE = 0;

            P1_POINTS = 99;

            P2_POINTS = 99;

            CURRENT_ROUND = 1;

            firstPlayer = 1;

            isGameOver = false;

            startNewRound();

        }

        function startNewRound() {

            p1Bet = 0;

            p2Bet = 0;

            isFirstBet = true;

            currentPlayer = firstPlayer;

            $resultPopup.style.display = 'none';

            $passButton.style.display = 'none';

            $nextRoundButton.style.display = 'none';

            $restartButton.style.display = 'none';

            $passMessage.style.display = 'none';

            $betButton.style.display = 'block';

            $pointInput.style.display = 'block';

            updateUI();

        }

        // === 게임 로직 함수 ===

        function handleBet() {

            if (isGameOver) return;

            const betAmount = parseInt($pointInput.value);

            const currentPoints = (currentPlayer === 1) ? P1_POINTS : P2_POINTS;

            if (isNaN(betAmount) || betAmount < 1 || betAmount > currentPoints) {

                alert(`유효한 포인트(1~${currentPoints}P)를 입력하세요.`);

                return;

            }

            // 1. 포인트 차감 및 베팅 기록

            if (currentPlayer === 1) {

                p1Bet = betAmount;

                P1_POINTS -= betAmount;

            } else { // currentPlayer === 2

                p2Bet = betAmount;

                P2_POINTS -= betAmount;

            }

           

            // 2. 턴 변경

            $pointInput.value = '10'; // 기본값으로 리셋

            updateUI();

            if (isFirstBet) {

                isFirstBet = false;

                showPassTurnPopup();

            } else {

                // 두 번째 베팅 완료 -> 라운드 결과 처리

                processRoundResult();

            }

        }

        function showPassTurnPopup() {

            // P1 -> P2, 또는 P2 -> P1 전환

            const nextPlayer = (currentPlayer === 1) ? 2 : 1;

            currentPlayer = nextPlayer;

            $resultHeader.textContent = `P${nextPlayer} 턴 준비`;

            $roundResults.innerHTML = `P${nextPlayer}의 베팅을 위해 휴대폰을 전달하세요.`;

            $passMessage.style.display = 'block';

            $passButton.style.display = 'block';

            $betButton.style.display = 'none';

            $pointInput.style.display = 'none';

            $resultPopup.style.display = 'block';

        }

        function nextPlayerTurn() {

            // 팝업 닫고 게임 재개

            $resultPopup.style.display = 'none';

            $passButton.style.display = 'none';

            $passMessage.style.display = 'none';

            $betButton.style.display = 'block';

            $pointInput.style.display = 'block';

            updateUI(); // 새로운 플레이어의 턴 정보 표시

        }

        function getBetColor(bet) {

            return (bet >= 1 && bet <= 9) ? '검은색' : '흰색';

        }

       

        function getColorClass(color) {

            return (color === '검은색') ? 'color-black' : 'color-white';

        }

        function processRoundResult() {

            let roundWinner = 0; // 0: 무승부, 1: P1, 2: P2

            if (p1Bet > p2Bet) {

                roundWinner = 1;

                P1_SCORE++;

                firstPlayer = 1;

            } else if (p2Bet > p1Bet) {

                roundWinner = 2;

                P2_SCORE++;

                firstPlayer = 2;

            }

            // else: 무승부 (0), 선 플레이어 유지 (firstPlayer 변동 없음)

            const p1Color = getBetColor(p1Bet);

            const p2Color = getBetColor(p2Bet);

            const p1Class = getColorClass(p1Color);

            const p2Class = getColorClass(p2Color);

           

            // 결과 메시지 생성

            let resultText = `

                <p>P1 베팅: <span class="${p1Class}">${p1Bet}P (${p1Color})</span></p>

                <p>P2 베팅: <span class="${p2Class}">${p2Bet}P (${p2Color})</span></p>

                <hr style="border-top: 1px solid #eee;">

            `;

            if (roundWinner === 0) {

                resultText += `<p>결과: **무승부!** (선 플레이어 P${firstPlayer} 유지)</p>`;

            } else {

                resultText += `<p>결과: **P${roundWinner} 승리!** (다음 선 플레이어: P${roundWinner})</p>`;

            }

            showRoundResult(resultText, roundWinner);

        }

        function showRoundResult(resultText, roundWinner) {

            $resultHeader.textContent = `라운드 ${CURRENT_ROUND} 결과`;

            $roundResults.innerHTML = resultText;

            $betButton.style.display = 'none';

            $pointInput.style.display = 'none';

           

            // 최종 승리 판정

            if (P1_SCORE >= WIN_SCORE || P2_SCORE >= WIN_SCORE || CURRENT_ROUND >= MAX_ROUNDS) {

                endGame();

            } else {

                $nextRoundButton.style.display = 'block';

            }

            $resultPopup.style.display = 'block';

        }

        function nextRound() {

            CURRENT_ROUND++;

            startNewRound();

        }

        function endGame() {

            isGameOver = true;

            $nextRoundButton.style.display = 'none';

            $restartButton.style.display = 'block';

            let finalMessage = "게임 종료!";

            if (P1_SCORE > P2_SCORE) {

                finalMessage = `👑 P1 최종 승리! (${P1_SCORE} vs ${P2_SCORE})`;

            } else if (P2_SCORE > P1_SCORE) {

                finalMessage = `👑 P2 최종 승리! (${P2_SCORE} vs ${P1_SCORE})`;

            } else {

                finalMessage = `무승부! (${P1_SCORE} vs ${P2_SCORE})`;

            }

            $resultHeader.textContent = finalMessage;

            $roundResults.innerHTML += `<p style="margin-top:15px; font-weight:bold;">총 9라운드 또는 5점 선취 게임 종료.</p>`;

        }

        function restartGame() {

            initGame();

        }

        // === 이벤트 리스너 ===

        $betButton.addEventListener('click', handleBet);

        $pointInput.addEventListener('keydown', (e) => {

            if (e.key === 'Enter') {

                handleBet();

            }

        });

       

        // === 게임 시작 ===

        initGame();
